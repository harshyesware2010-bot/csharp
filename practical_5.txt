using arithmeticmvc.Models;
using Microsoft.AspNetCore.Mvc;
using System.Diagnostics;
namespace arithmeticmvc.Controllers{
public class HomeController : Controller
{ public IActionResult Index(Calculator calc){
switch (calc.operation){
case "+":
calc.Result = calc.Number1 + calc.Number2;
break;
case "-":
calc.Result = calc.Number1 - calc.Number2;
break;
case "*":
calc.Result = calc.Number1 * calc.Number2;
break;
case "/":
if (calc.Number2 != 0)
{
calc.Result = calc.Number1 / calc.Number2;
}
break;} return View(calc);}}}

1. Create Project
Step 1: Open Visual Studio 2026
Step 2: Create → `ASP.NET Web Application (.NET Framework)
Step 3: Select: `Empty` and `Check Web Forms`
Step 4: Click Create

2. Add Web Form
Right-click Project → `Add` → `Web Form`
Name: `CookieDemo.aspx`
3.
Add Controls Label Text|Control Type|	ID
User Name	         TextBox	txtName
Save Cookie	         Button	        btnSave
Show Cookies	         Button	        btnShow
Output	                 Label	        lblOutput


Write Code for Save Cookie Button

protected void btnSave_Click(object sender, EventArgs e) {
HttpCookie sessionCookie = new HttpCookie("SessionUser");
sessionCookie.Value = txtName.Text;
Response.Cookies.Add(sessionCookie);
HttpCookie persistentCookie = new HttpCookie("PersistentUser");
persistentCookie.Value = txtName.Text;
persistentCookie.Expires = DateTime.Now.AddDays(7);
Response.Cookies.Add(persistentCookie);
lblOutput.Text = "Cookies Saved Successfully";} }


Write Code for Show Cookie Button

protected void btnShow_Click(object sender, EventArgs e)
{
string sessionData = Request.Cookies["SessionUser"].Value;
string persistentData = Request.Cookies["PersistentUser"].Value;
lblOutput.Text = "Session Cookie: " + sessionData +
"<br/>Persistent Cookie: " + persistentData;
}
